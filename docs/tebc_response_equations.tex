\section*{Overview}

\paragraph{Cell categories.} In the independent and place-dependent models, every
neuron $i$ falls into exactly one of four categories, determined by two independent
boolean draws: whether it carries a place field
(probability \texttt{percent\_place\_cells}) and whether it is TEBC-responsive
(probability from \texttt{percent\_task\_responsive\_cells}):

\begin{center}
\begin{tabular}{lll}
    \hline
    Category & Place field? & TEBC-responsive? \\
    \hline
    place-only & yes & no \\
    place+task & yes & yes \\
    task-only  & no  & yes \\
    silent     & no  & no \\
    \hline
\end{tabular}
\end{center}

\noindent
Only \textbf{place+task} cells blend a task and a place signal; place-only cells fire
purely off the place signal, task-only cells fire purely off the task signal, and
silent cells sit at a fixed near-zero baseline. The arousal-mediated model has no such
categories: every cell is a place cell, and its task response (if TEBC-responsive) is
selected by the animal's instantaneous running/in-field state instead (see \S5).

\paragraph{Balance parameter $\beta_i$.}
Each \emph{place+task} neuron $i$ has a balance parameter $\beta_i \in [0,1]$ drawn from
a population distribution $P_{\mathrm{balance}}$, controlling the relative contribution
of the task-driven signal versus the place-cell signal. $\beta_i = 1$ means the neuron's
task signal fully dominates; $\beta_i = 0$ means it behaves as a pure place cell.
$\beta_i$ has no effect on place-only, task-only, or silent cells.

\paragraph{Model summary (place+task category only).}

\medskip
\begin{center}
\begin{tabular}{p{2.8cm}p{6.5cm}p{5cm}}
    \hline
    Model & High-level equation & Description \\
    \hline
    Independent &
        $r_i = \beta_i\, r^{\mathrm{task}}_i(t) + (1-\beta_i)\tilde{r}^{\mathrm{place}}_i + \varepsilon_i$ &
        The task signal is independent of the animal's current location and determined strictly as a function of the time since the onset of the stimuli. Each task-influenced neuron is randomly assigned 1 of 8 response functions, always scaled to a fixed peak amplitude. \\[8pt]
    Place-Dependent &
        $r_i = \beta_i\, r^{\mathrm{task}}_i(t,\tilde{r}^{\mathrm{place}}_i) + (1-\beta_i)\tilde{r}^{\mathrm{place}}_i + \varepsilon_i$ &
        The task signal is dependent on the animal's current location: it is scaled to the neuron's own live place rate rather than a fixed amplitude. Each task-influenced neuron is randomly assigned 1 of 8 response functions. \\[8pt]
    Arousal-Mediated &
        $r_i = \tilde{r}^{\mathrm{place}}_i + r^{\mathrm{task}}_i(t,\tilde{r}^{\mathrm{place}}_i) + \varepsilon_i$ &
        The task response is selected from one of 4 functions based on whether the rat is running and whether the cell is in a place field, and is additive on top of the place rate rather than blended with it. \\
    \hline
\end{tabular}
\end{center}

\paragraph{Key symbols.}
\begin{itemize}
    \item $\tilde{r}^{\mathrm{place}}_i$ --- velocity-modulated Gaussian place cell rate
    \item $r^{\mathrm{task}}_i(t)$ --- TEBC task response for neuron $i$, drawn from type $k \in \{1,\dots,8\}$ (independent/place-dependent) or $k \in \{1,\dots,4\}$ (arousal-mediated)
    \item $A_i$ --- per-neuron task response \emph{peak amplitude}: the ceiling the normalized response shape (see \S4) is rescaled to. Model-specific; see \S2/\S3.
    \item $\beta_i$ --- per-neuron balance parameter (place+task cells only)
    \item $b_i \equiv \tilde{r}^{\mathrm{place}}_i(t)$ --- live place rate, used as the task amplitude $A_i$ in the place-dependent and arousal-mediated models
    \item $\varepsilon_i \sim \mathcal{N}\!\left(-\tfrac{0.02}{30},\, \tfrac{0.02}{30}\right)$ --- additive noise term, drawn independently per neuron per timestep, common to all models
\end{itemize}

\paragraph{Notation.} All firing rates are in Hz. Time $t$ is seconds since CS onset.
Constants: $t_{CS} = 0.25\,\text{s}$, $t_{US} = 0.75\,\text{s}$, $d_{US} = 0.1\,\text{s}$.

\section{Place Cell Firing Rate (RatInABox)}

All models inherit the Gaussian place cell implementation from the RatInABox library.
For each neuron $i$ with field centre $\mathbf{c}_i$ and width $\sigma$:

\begin{equation}
    r^{\mathrm{PC}}_i(\mathbf{x}) =
        \exp\!\left(-\frac{d(\mathbf{x},\,\mathbf{c}_i)^2}{2\sigma^2}\right)
        \cdot (f_{\max} - f_{\min}) + f_{\min}
\end{equation}

\noindent
where $d(\mathbf{x}, \mathbf{c}_i)$ is the geodesic distance from the agent position
$\mathbf{x}$ to the place field centre and $\sigma = 0.20\,\text{m}$.
$f_{\min} = 0\,\text{Hz}$ and $f_{\max} = 12\,\text{Hz}$ for \emph{all three} models ---
the independent and place-dependent models share the same underlying place-cell
construction (\texttt{CombinedPlaceTebc}), so independent no longer uses a reduced
$f_{\max}=1\,\text{Hz}$ ceiling.

% ---------------------------------------------------------------------------
\section{Independent Model}

\textit{The task signal is independent of the animal's current location and determined
strictly as a function of the time since the onset of the stimuli. Each task-influenced
neuron is randomly assigned 1 of 8 response functions, and the task amplitude is a
fixed constant shared by every task-responsive neuron.}

\bigskip\noindent
\textbf{place-only cells:}
\begin{equation}
    r_i(t) = \tilde{r}^{\mathrm{place}}_i(t) + \varepsilon_i
\end{equation}

\noindent
\textbf{place+task cells} (in trial; falls back to the place-only equation outside a trial):
\begin{equation}
    r_i(t) = \beta_i\, r^{\mathrm{task}}_i(t)
           + \bigl(1 - \beta_i\bigr)\, \tilde{r}^{\mathrm{place}}_i(t)
           + \varepsilon_i
\end{equation}

\noindent
\textbf{task-only cells} (in trial; falls back to baseline outside a trial):
\begin{equation}
    r_i(t) = r^{\mathrm{task}}_i(t) + \varepsilon_i
\end{equation}

\noindent
\textbf{silent cells:}
\begin{equation}
    r_i(t) = \tfrac{0.02}{30} + \varepsilon_i
\end{equation}

\noindent
In all cases $\varepsilon_i \sim \mathcal{N}\!\left(-\tfrac{0.02}{30}, \tfrac{0.02}{30}\right)$
drawn independently per neuron, and $\beta_i \in [0, 1]$ is drawn from
$\{\beta_i\} \sim P_{\mathrm{balance}}$.

\paragraph{Velocity-modulated place cell rate.}
Let $r^{\mathrm{PC}}_i$ be the standard Gaussian place cell firing rate and $v$ the
agent's smoothed speed (m/s). The modulated rate is:

\begin{equation}
    \tilde{r}^{\mathrm{place}}_i =
    \begin{cases}
        \tfrac{0.02}{30} & v < 0.02 \\[4pt]
        \dfrac{p(100v)}{30}\; r^{\mathrm{PC}}_i & v \geq 0.02
    \end{cases}
\end{equation}

\noindent
where $p$ is the empirical cubic polynomial

\begin{equation}
    p(s) = -3.261 \times 10^{-4}\, s^3
          + 1.741 \times 10^{-2}\, s^2
          + 8.366 \times 10^{-2}\, s
          + 1.161
\end{equation}

\noindent
(argument $s = 100v$ converts m/s to cm/s).
Non-place-cell neurons have their rate fixed at $\tfrac{0.02}{30}$ regardless of $v$.

\paragraph{Task firing rate.}
\begin{equation}
    r^{\mathrm{task}}_i(t) =
    \begin{cases}
        R_{\,c_i}(t;\, A_i) & \text{neuron } i \text{ is TEBC-responsive} \\
        0            & \text{otherwise}
    \end{cases}
\end{equation}

\noindent
where $c_i \in \{1,\dots,8\}$ is the cell type assigned to neuron $i$ and $A_i$ is the
peak amplitude every response profile is rescaled to (\S4). In the independent model
\textbf{$A_i$ is the same fixed constant for every task-responsive neuron}
(place+task and task-only alike): $A_i = 12\,\text{Hz}$, the shared population $f_{\max}$
(\S1) --- i.e. the response is not scaled by the individual cell's own place rate.
The response profiles $R_k(t; A)$ map to the equations in \S4 as follows:

\medskip
\begin{center}
\begin{tabular}{cll}
    \hline
    $k$ & Description & Equation \\
    \hline
    1 & Bimodal Gaussian          & \eqref{eq:type1} \\
    2 & One-sided Gaussian        & \eqref{eq:type2} \\
    3 & Piecewise Linear Decay    & \eqref{eq:type3} \\
    4 & Ramp with Gaussian Burst  & \eqref{eq:type4} \\
    5 & Trapezoidal US Response   & \eqref{eq:type5} \\
    6 & Noisy Uniform             & \eqref{eq:type6} \\
    7 & Windowed Bimodal Gaussian & \eqref{eq:type7} \\
    8 & Post-CS Piecewise Decay   & \eqref{eq:type8} \\
    \hline
\end{tabular}
\end{center}

\section{Place-Dependent Model}

\textit{The task signal is dependent on the animal's current location: rather than a
fixed constant, its peak amplitude is the neuron's own live place rate. Each
task-influenced neuron is randomly assigned 1 of 8 response functions.}

\bigskip\noindent
\textbf{place-only cells:}
\begin{equation}
    r_i(t) = \tilde{r}^{\mathrm{place}}_i(t) + \varepsilon_i
\end{equation}

\noindent
\textbf{place+task cells} (in trial; falls back to the place-only equation outside a trial):
\begin{equation}
    r_i(t) = \beta_i\, r^{\mathrm{task}}_i\!\bigl(t,\,\tilde{r}^{\mathrm{place}}_i(t)\bigr)
           + \bigl(1 - \beta_i\bigr)\, \tilde{r}^{\mathrm{place}}_i(t)
           + \varepsilon_i
\end{equation}

\noindent
\textbf{task-only cells} (in trial; falls back to baseline outside a trial):
\begin{equation}
    r_i(t) = r^{\mathrm{task}}_i(t) + \varepsilon_i
\end{equation}

\noindent
\textbf{silent cells:}
\begin{equation}
    r_i(t) = \tfrac{0.02}{30} + \varepsilon_i
\end{equation}

\noindent
$\varepsilon_i$ and $\beta_i$ are as defined in \S2. Let $b_i \equiv \tilde{r}^{\mathrm{place}}_i(t)$
be the neuron's own live velocity-modulated place rate:

\begin{equation}
    r^{\mathrm{task}}_i(t) =
    \begin{cases}
        R_{c_i}\!\bigl(t;\; A_i\bigr) & \text{neuron } i \text{ is TEBC-responsive} \\
        0 & \text{otherwise}
    \end{cases}
\end{equation}

\begin{equation}
    A_i =
    \begin{cases}
        b_i & \text{place+task cell (has a place field)} \\
        0.02/30 & \text{task-only cell (no place field)}
    \end{cases}
\end{equation}

\noindent
i.e. a place+task neuron's task response peaks at its own live place rate; a task-only
neuron (no place field to scale off) instead uses a small fixed placeholder amplitude.
Each profile $R_k(t; A)$ has the same functional form as in the independent model
(Equations~\eqref{eq:type1}--\eqref{eq:type8}); the shape-construction magnitude
parameters below are expressed as multiples of a unit baseline, matching the ratios
the shape is built with before renormalization (\S4):

\medskip
\begin{center}
\begin{tabular}{clll}
    \hline
    $k$ & Description & Shape magnitude ratios & Eq. \\
    \hline
    1 & Bimodal Gaussian         & $A_1 = 2.14$,\quad $A_2 = 5.71$                       & \eqref{eq:type1} \\
    2 & One-sided Gaussian       & $A = 3.14$                                             & \eqref{eq:type2} \\
    3 & Piecewise Linear Decay   & $d_1 = 0.33$,\quad $d_2 = 0.17$                       & \eqref{eq:type3} \\
    4 & Ramp with Gaussian Burst & $A_{CS} = 0.40$,\quad $A_{US} = 1.10$                 & \eqref{eq:type4} \\
    5 & Trapezoidal US Response  & $A_{US} = 3.14$                                        & \eqref{eq:type5} \\
    6 & Noisy Uniform            & (unchanged --- not renormalized)                       & \eqref{eq:type6} \\
    7 & Windowed Bimodal Gaussian& $A_1 = 1.67$,\quad $A_2 = 2.14$                       & \eqref{eq:type7} \\
    8 & Post-CS Piecewise Decay  & steps at $0.46$,\; $0.62$,\; $0.38$                   & \eqref{eq:type8} \\
    \hline
\end{tabular}
\end{center}

\paragraph{Velocity modulation.}
Identical to the independent model (same cubic polynomial $p(s)$ and the same
$\tfrac{0.02}{30}$ floor below the speed threshold):

\begin{equation}
    \tilde{r}^{\mathrm{place}}_i =
    \begin{cases}
        \tfrac{0.02}{30} & v < 0.02 \\[4pt]
        \dfrac{p(100v)}{30}\; r^{\mathrm{PC}}_i & v \geq 0.02
    \end{cases}
\end{equation}

\section{TEBC Response Profiles (Types 1--8)}

The following profiles are shared by the independent and place-dependent models.
Each type is built as a raw shape $S_k(t)$ using the magnitude constants given below
(fitted assuming a unit baseline/amplitude of $1$), then \textbf{renormalized} to run
from $0$ up to the neuron's actual peak amplitude $A_i$ (\S2/\S3):

\begin{equation}\label{eq:normalize}
    R_k(t; A) = \frac{S_k(t) - \min_\tau S_k(\tau)}{\max_\tau S_k(\tau) - \min_\tau S_k(\tau)} \cdot A,
    \qquad \tau \in [-0.1,\, 2.0]\,\text{s (1000-point grid)}
\end{equation}

\noindent
\textbf{Type 6 is the one exception} --- it is \emph{not} renormalized; see
\eqref{eq:type6}. The raw shapes $S_k(t)$ below use the magnitude constants exactly
as coded (independent model's fixed defaults; place-dependent's ratios from the table
in \S3), but only their \emph{relative} shape survives \eqref{eq:normalize} --- the
absolute constants shown do not appear directly in the final firing rate.

% ---------------------------------------------------------------------------
\subsection*{Type 1 --- Bimodal Gaussian}

\begin{equation}\label{eq:type1}
    S_1(t) = A_1 \exp\!\left(-\frac{(t-\mu_1)^2}{2\sigma_1^2}\right)
         + A_2 \exp\!\left(-\frac{(t-\mu_2)^2}{2\sigma_2^2}\right)
\end{equation}

\noindent
Independent model constants: $A_1 = 0.15$, $\mu_1 = 0.3$, $\sigma_1 = 0.1$;
$A_2 = 0.40$, $\mu_2 = 0.85$, $\sigma_2 = 0.1$; baseline $= 0.07$ (added into both
Gaussian terms before normalizing). Place-dependent model uses $A_1=2.14$, $A_2=5.71$,
same $\mu,\sigma$, baseline $=1.0$.

% ---------------------------------------------------------------------------
\subsection*{Type 2 --- One-sided Gaussian}

\begin{equation}\label{eq:type2}
    S_2(t) = \begin{cases}
        A \exp\!\left(-\dfrac{(t - \mu)^2}{2\sigma^2}\right) + b & t > \mu - 0.1 \\[6pt]
        b & \text{otherwise}
    \end{cases}
\end{equation}

\noindent
Independent: $A = 0.22$, $\mu = 0.75$, $\sigma = 0.15$, $b = 0.07$.
Place-dependent: $A = 3.14$, same $\mu,\sigma$, $b = 1.0$.

% ---------------------------------------------------------------------------
\subsection*{Type 3 --- Piecewise Linear Decay}

Independent: $b = 0.06$, $d_1 = 0.02$, $d_2 = 0.01$.
Place-dependent: $b = 1.0$, $d_1 = 0.33$, $d_2 = 0.17$.

\begin{equation}\label{eq:type3}
    S_3(t) = \begin{cases}
        b                                                              & t \leq 0            \\[4pt]
        \dfrac{d_1 - b}{0.2}\,t + b                                   & 0 < t < 0.2         \\[8pt]
        \dfrac{d_2 - d_1}{0.65}(t - 0.2) + d_1                      & 0.2 \leq t < 0.85   \\[6pt]
        d_2                                                            & 0.85 \leq t \leq 3.85 \\[4pt]
        b                                                              & t > 3.85
    \end{cases}
\end{equation}

% ---------------------------------------------------------------------------
\subsection*{Type 4 --- Linear Ramp with Gaussian Burst}

Independent: $b = 0.075$, $A_{CS} = 0.03$, $A_{US} = 0.08$, $\sigma_{US} = 0.2$.
Place-dependent: $b = 1.0$, $A_{CS} = 0.40$, $A_{US} = 1.10$, $\sigma_{US} = 0.2$.
Define the unclamped response:

\begin{equation}
    \tilde{S}_4(t) = \begin{cases}
        b                                                                                     & t \leq 0.05          \\[4pt]
        \dfrac{A_{CS} - b}{0.73}(t - 0.05) + b                                              & 0.05 < t \leq 0.78   \\[8pt]
        A_{US}\exp\!\left(-\dfrac{(t - 0.78)^2}{5\,\sigma_{US}^5}\right) + b               & 0.78 < t \leq 1      \\[6pt]
        A_{CS}                                                                               & 1 < t \leq 3         \\[4pt]
        b                                                                                     & t > 3
    \end{cases}
\end{equation}

The output is clamped to $[A_{CS},\, A_{US}]$ before normalizing:
\begin{equation}\label{eq:type4}
    S_4(t) = \min\!\bigl(A_{US},\; \max\!\bigl(A_{CS},\; \tilde{S}_4(t)\bigr)\bigr)
\end{equation}

% ---------------------------------------------------------------------------
\subsection*{Type 5 --- Trapezoidal US Response}

\begin{equation}\label{eq:type5}
    S_5(t) = \begin{cases}
        b                                                                    & t < 0.75            \\[4pt]
        A_{US}                                                               & 0.75 \leq t \leq 0.78 \\[4pt]
        A_{US} + \dfrac{b - A_{US}}{0.42}(t - 0.78)                        & 0.78 < t < 1.2      \\[8pt]
        b                                                                    & t \geq 1.2
    \end{cases}
\end{equation}

\noindent
Independent: $b = 0.035$, $A_{US} = 0.11$. Place-dependent: $b = 1.0$, $A_{US} = 3.14$.

% ---------------------------------------------------------------------------
\subsection*{Type 6 --- Noisy Uniform (not renormalized)}

\begin{equation}\label{eq:type6}
    R_6(t; A) = 0.5\,A+\varepsilon, \qquad \varepsilon \sim \mathcal{N}\!\left(-\eta,\; \eta\right), \quad \eta = \frac{0.03}{30}
\end{equation}

\noindent
Unlike types 1--5 and 7--8, this profile is applied directly to the peak amplitude
$A$ (not passed through \eqref{eq:normalize}).

% ---------------------------------------------------------------------------
\subsection*{Type 7 --- Windowed Bimodal Gaussian}

Each Gaussian peak is truncated to a one-sided window:

\begin{equation}
    G_i(t) = \begin{cases}
        A_i \exp\!\left(-\dfrac{(t - \mu_i)^2}{2\sigma_i^2}\right) & \mu_i - 3\sigma_i \leq t \leq \mu_i + D_i \\[6pt]
        0 & \text{otherwise}
    \end{cases}
\end{equation}

The raw shape is:

\begin{equation}\label{eq:type7}
    S_7(t) = \max\!\left(\sum_{i=1}^{2} G_i(t) + b,\; b\right)
\end{equation}

\noindent
Independent: $A_1 = 0.07$, $\mu_1 = 0.30$, $\sigma_1 = 0.08$, $D_1 = 0.25$;
$A_2 = 0.09$, $\mu_2 = 0.80$, $\sigma_2 = 0.15$, $D_2 = 0.10$; $b = 0.042$.
Place-dependent: $A_1 = 1.67$, $A_2 = 2.14$, same $\mu,\sigma,D$, $b = 1.0$.

% ---------------------------------------------------------------------------
\subsection*{Type 8 --- Post-CS Piecewise Decay}

Independent: $b = 0.065$, with fixed absolute steps $0.03$, $0.04$, $0.025$.
Place-dependent: $b = 1.0$, with steps expressed as $0.46\,b$, $0.62\,b$, $0.38\,b$.

\begin{equation}\label{eq:type8}
    S_8(t) = \begin{cases}
        b                                                                        & t \leq 0.25              \\[4pt]
        s_1                                                                      & 0.25 < t \leq 0.252      \\[6pt]
        \dfrac{s_2 - b}{0.498}(t - 0.252) + b                                  & 0.252 < t < 0.75         \\[8pt]
        \dfrac{s_3 - b}{0.75}(t - 0.75) + b                                    & 0.75 \leq t \leq 1.5     \\[6pt]
        b                                                                        & t > 1.5
    \end{cases}
\end{equation}

\noindent
where $(s_1, s_2, s_3) = (0.03, 0.04, 0.025)$ for the independent model, and
$(s_1,s_2,s_3) = (0.46b,\, 0.62b,\, 0.38b)$ for the place-dependent model.

\section{Arousal-Mediated Model}

\textit{The task response is selected from one of 4 functions based on whether
the rat is running and whether the cell is in a place field. Each of these functions
depends solely on the time since CS onset and the neuron's own live place rate.}

\bigskip\noindent
There is no balance parameter and no place/task category split --- every neuron is a
place cell, and the task response (if TEBC-responsive) is added directly on top:

\begin{equation}
    r_i(t) = \tilde{r}^{\mathrm{place}}_i(t) + r^{\mathrm{task}}_i(t) + \varepsilon_i,
    \qquad \varepsilon_i \sim \mathcal{N}\!\left(-\tfrac{0.02}{30},\; \tfrac{0.02}{30}\right)
\end{equation}

\paragraph{Velocity modulation.}
Uses the same cubic polynomial $p(s)$ as \S2, with a different slow-speed floor. A
randomly chosen subset of neurons, of size $(1-\texttt{percent\_place\_cells})\times N$,
is additionally forced to a near-zero rate whenever the animal is running --- \emph{but
not} when it is stopped, so this subset only reads as ``non-place'' during locomotion:

\begin{equation}
    \tilde{r}^{\mathrm{place}}_i =
    \begin{cases}
        \tfrac{0.2}{30}\; r^{\mathrm{PC}}_i & v < 0.02 \\[4pt]
        \dfrac{p(100v)}{30}\; r^{\mathrm{PC}}_i & v \geq 0.02,\ i \notin \mathcal{Z} \\[4pt]
        \tfrac{0.02}{30} & v \geq 0.02,\ i \in \mathcal{Z}
    \end{cases}
\end{equation}

\noindent
where $\mathcal{Z}$ is the randomly sampled zeroed-out index set (resampled once per
simulation, not per timestep).

\paragraph{Task firing rate.}
The response function is selected by two binary conditions evaluated each timestep:
whether the neuron is inside its place field, and whether the animal is running.
Place-field membership is assessed from the \emph{raw} (pre-velocity-modulation)
place rate $r^{\mathrm{PC}}_i$:

\begin{equation}
    \text{in\_field}_i = \bigl[r^{\mathrm{PC}}_i \geq 0.2\bigr], \qquad
    \text{running} = \bigl[v > 0.02\bigr]
\end{equation}

\begin{equation}
    r^{\mathrm{task}}_i(t) =
    \begin{cases}
        0 & \text{neuron } i \text{ not TEBC-responsive} \\[4pt]
        R_1\!\bigl(t,\, b_i\bigr) & \text{in field},\quad \text{running} \\[4pt]
        R_2\!\bigl(t,\, b_i\bigr) & \text{out of field},\quad \text{running} \\[4pt]
        R_3\!\bigl(t,\, b_i\bigr) & \text{in field},\quad \text{stopped} \\[4pt]
        R_4\!\bigl(t,\, b_i\bigr) & \text{out of field},\quad \text{stopped}
    \end{cases}
\end{equation}

\noindent
where $b_i = \tilde{r}^{\mathrm{place}}_i(t)$ is the velocity-modulated place rate
used as the baseline for all magnitude expressions.

\paragraph{Response functions.}

\medskip\noindent
\textbf{$R_1$ --- In field, running.} Elevated through the CS and trace interval
(the hippocampal ``bridging'' signature expected of a trace-conditioned cell), then
peaks more strongly during the US pulse itself before returning to baseline:
\begin{equation}
    R_1(t, b) = \begin{cases}
        1.3\,b  & t < t_{US}                        \\[4pt]
        1.5\,b  & t_{US} \leq t < t_{US} + d_{US}    \\[4pt]
        b       & t \geq t_{US} + d_{US}
    \end{cases}
\end{equation}

\medskip\noindent
\textbf{$R_2$ --- Out of field, running:} a slight, sustained increase through the
CS, trace interval, and US pulse:
\begin{equation}
    R_2(t, b) = \begin{cases}
        1.1\,b  & t < t_{US} + d_{US}   \\[4pt]
        b       & t \geq t_{US} + d_{US}
    \end{cases}
\end{equation}

\medskip\noindent
\textbf{$R_3$ --- In field, stopped:}
\begin{equation}
    R_3(t, b) = \begin{cases}
        1.5\,b                                                               & t < t_{CS}              \\[6pt]
        b\!\left(1 + 0.5\,\dfrac{t_{US} - t}{t_{US} - t_{CS}}\right)       & t_{CS} \leq t < t_{US}  \\[8pt]
        3\,b                                                                 & t_{US} \leq t < t_{US} + d_{US} \\[4pt]
        b                                                                    & t \geq t_{US} + d_{US}
    \end{cases}
\end{equation}

\medskip\noindent
\textbf{$R_4$ --- Out of field, stopped:}
\begin{equation}
    R_4(t, b) = b + \varepsilon_4, \qquad \varepsilon_4 \sim \mathrm{Uniform}(-0.2\,b,\; 0.2\,b)
\end{equation}
